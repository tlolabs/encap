cmake_minimum_required(VERSION 3.20)
project(encap_desktop_adapters LANGUAGES C CXX)
add_library(encap-playback SHARED playback.c)
set_target_properties(encap-playback PROPERTIES C_STANDARD 17 C_VISIBILITY_PRESET hidden)
if(APPLE)
  target_link_libraries(encap-playback PRIVATE "-framework CoreAudio" "-framework AudioToolbox" "-framework CoreFoundation")
elseif(WIN32)
  target_link_libraries(encap-playback PRIVATE ole32)
else()
  target_link_libraries(encap-playback PRIVATE m pthread dl)
endif()

if(WIN32)
  add_library(encap-media SHARED media_windows.cpp)
  target_compile_features(encap-media PRIVATE cxx_std_17)
  target_link_libraries(encap-media PRIVATE windowsapp)
elseif(UNIX AND NOT APPLE)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0)
  add_library(encap-media SHARED media_linux.c)
  target_link_libraries(encap-media PRIVATE PkgConfig::GIO)
  set_target_properties(encap-media PROPERTIES C_STANDARD 17 C_VISIBILITY_PRESET hidden)
endif()
